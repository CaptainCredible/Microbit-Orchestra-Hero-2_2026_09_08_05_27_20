//Set to 2.0 for AA
#define AA 1.0
#define AA_INT 1    // keep in sync with AA

#define STEPS 80.0
#define MDIST 35.0
#define pi 3.1415926535
#define rot(a) mat2(cos(a),sin(a),-sin(a),cos(a))
#define sat(a) clamp(a,0.0,1.0)

#define ITERS_TRACE 9
#define ITERS_NORM 20

#define HOR_SCALE 1.1
#define OCC_SPEED 1.4
#define DX_DET 0.65

#define FREQ 0.6
#define HEIGHT_DIV 2.5
#define WEIGHT_SCL 0.8
#define FREQ_SCL 1.2
#define TIME_SCL 1.095
#define WAV_ROT 1.21
#define DRAG 0.9
#define SCRL_SPEED 5.5
const vec2 scrollDir = vec2(0.0, 1.0);

// ---- Camera controls -------------------------------------------------------
#define CAM_PITCH   1.0    // radians; positive tilts the view up
#define CAM_HEIGHT  6.475  // camera Y; the original is 2.25 * 1.1
#define FLY_SPEED   0.0    // world units per second along +z; 0 = stationary
// ---------------------------------------------------------------------------

// ---- Colour controls -------------------------------------------------------
// The whole scene derives from one cosine palette:
//     spc(n, bright) = PAL_A*bright + PAL_B*cos(2*pi*(PAL_C*n + PAL_D))
// Change these and every colour moves together.
#define PAL_A   vec3(1.0)               // base brightness
#define PAL_B   vec3(0.5)               // contrast
#define PAL_C   vec3(1.0)               // frequency
#define PAL_D   vec3(0.0, -0.5, -0.5)   // phase - shifts the hue

// Where on the palette the scene sits. 0.13 is the original. Nudge it to
// slide the whole scene through the palette.
#define SPEC    0.13

// Per-element tints. 1.0 means "leave it as the palette gave it".
#define SKY_TINT    vec3(1.0)
#define SUN_TINT    vec3(1.0)
#define WATER_TINT  vec3(1.0)

// How strongly the palette colours the water. 0.4 is the original.
#define WATER_COLOR 0.4
// ---------------------------------------------------------------------------

vec2 wavedx(vec2 wavPos, float t){
    vec2 dx = vec2(0.0);
    vec2 wavDir = vec2(1.0, 0.0);
    float wavWeight = 1.0;
    wavPos += t * SCRL_SPEED * scrollDir;
    wavPos *= HOR_SCALE;
    float wavFreq = FREQ;
    float wavTime = OCC_SPEED * t;
    for (int i = 0; i < ITERS_NORM; i++){
        wavDir *= rot(WAV_ROT);
        float x = dot(wavDir, wavPos) * wavFreq + wavTime;
        float result = exp(sin(x) - 1.0) * cos(x);
        result *= wavWeight;
        dx += result * wavDir / pow(wavWeight, DX_DET);
        wavFreq *= FREQ_SCL;
        wavTime *= TIME_SCL;
        wavPos -= wavDir * result * DRAG;
        wavWeight *= WEIGHT_SCL;
    }
    float wavSum = -(pow(WEIGHT_SCL, float(ITERS_NORM)) - 1.0) * HEIGHT_DIV;
    return dx / pow(wavSum, 1.0 - DX_DET);
}

float wave(vec2 wavPos, float t){
    float wav = 0.0;
    vec2 wavDir = vec2(1.0, 0.0);
    float wavWeight = 1.0;
    wavPos += t * SCRL_SPEED * scrollDir;
    wavPos *= HOR_SCALE;
    float wavFreq = FREQ;
    float wavTime = OCC_SPEED * t;
    for (int i = 0; i < ITERS_TRACE; i++){
        wavDir *= rot(WAV_ROT);
        float x = dot(wavDir, wavPos) * wavFreq + wavTime;
        float w = exp(sin(x) - 1.0) * wavWeight;
        wav += w;
        wavFreq *= FREQ_SCL;
        wavTime *= TIME_SCL;
        wavPos -= wavDir * w * DRAG * cos(x);
        wavWeight *= WEIGHT_SCL;
    }
    float wavSum = -(pow(WEIGHT_SCL, float(ITERS_TRACE)) - 1.0) * HEIGHT_DIV;
    return wav / wavSum;
}

vec3 norm(vec3 p){
    vec2 wav = -wavedx(p.xz, iTime);
    return normalize(vec3(wav.x, 1.0, wav.y));
}

float map(vec3 p){
    p.y -= wave(p.xz, iTime);
    return p.y;
}

// Cosine palette. PAL_A, PAL_B, PAL_C, PAL_D are the classic four parameters.
vec3 pal(float t, float bright){
    return PAL_A * bright + PAL_B * cos(2.0 * pi * (PAL_C * t + PAL_D));
}

vec3 spc(float n, float bright){
    return pal(n, bright);
}

const vec2 sunrot = vec2(-0.3, -0.25);

vec3 sky(vec3 rd){
    float px = 1.5 / min(iResolution.x, iResolution.y);
    vec3 rdo = rd;
    float rad = 0.075;
    vec3 col = vec3(0.0);

    rd.yz *= rot(sunrot.y);
    rd.xz *= rot(sunrot.x);
    float sFade = 2.5 / min(iResolution.x, iResolution.y);
    float zFade = rd.z * 0.5 + 0.5;

    vec3 sc = spc(SPEC - 0.1, 0.6) * 0.85;
    float a = length(rd.xy);
    vec3 sun = smoothstep(a - px - sFade, a + px + sFade, rad) * sc * zFade * 2.0;
    col += sun * SUN_TINT;
    col += rad / (rad + pow(a, 1.7)) * sc * zFade;
    col = col + mix(col, spc(SPEC + 0.1, 0.8), sat(1.0 - length(col))) * 0.2;

    float e = 0.0;
    vec3 p = rdo;
    p.xz *= 0.4;
    p.x += iTime * 0.007;
    float s = 200.0;
    for (int k = 0; k < 14; k++){
        if (s <= 10.0) break;
        p.xz *= rot(s);
        p += s;
        e += abs(dot(sin(p * s + iTime * 0.02) / s, vec3(1.65)));
        s *= 0.8;
    }
    e *= smoothstep(0.5, 0.4, e - 0.095);

    col += (e) * smoothstep(-0.02, 0.3, rdo.y) * 0.8 * (1.0 - sun * 3.75) * mix(sc, vec3(1.0), 0.4);

    return col * SKY_TINT;
}

void render(out vec4 fragColor, in vec2 fragCoord){
    vec2 uv = (fragCoord - 0.5 * iResolution.xy) / min(iResolution.y, iResolution.x);
    vec3 col = vec3(0.0);

    vec3 ro = vec3(0.0, CAM_HEIGHT, -3.0 * 1.1);
    ro.z += iTime * FLY_SPEED;

    bool click = iMouse.z > 0.0;
    //if (click){
    //    ro.yz *= rot(2.0 * min(iMouse.y / iResolution.y - 0.5, 0.15));
    //    ro.zx *= rot(-7.0 * (iMouse.x / iResolution.x - 0.5));
    //}

    vec3 lk = vec3(0.0, 2.0, 0.0);
    lk.y += CAM_PITCH * 3.0;

    vec3 f = normalize(lk - ro);
    vec3 r = normalize(cross(vec3(0.0, 1.0, 0.0), f));
    vec3 rd = normalize(f * 0.9 + uv.x * r + uv.y * cross(f, r));

    float dO = 0.0;
    bool hit = false;
    float d = 0.0;
    vec3 p = ro;

    float tPln = -(ro.y - 1.86) / rd.y;
    if (tPln > 0.0 || click){
        if (!click) dO += tPln;
        for (float i = 0.0; i < STEPS; i++){
            p = ro + rd * dO;
            d = map(p);
            dO += d;
            if (abs(d) < 0.005 || i > STEPS - 2.0){
                hit = true;
                break;
            }
            if (dO > MDIST){
                dO = MDIST;
                break;
            }
        }
    }
    vec3 skyrd = sky(rd);
    if (hit){
        vec3 n = norm(p);
        vec3 rfl = reflect(rd, n);
        rfl.y = abs(rfl.y);
        vec3 rf = refract(rd, n, 1.0 / 1.33);
        float fres = clamp((pow(1.0 - max(0.0, dot(-n, rd)), 5.0)), 0.0, 1.0);

        vec3 sunDir = vec3(0.0, 0.15, 1.0);
        sunDir.xz *= rot(-sunrot.x);
        col += sky(rfl) * fres * 0.9;
        float subRefract = pow(max(0.0, dot(rf, sunDir)), 35.0);
        col += pow(spc(SPEC - 0.1, 0.5), vec3(2.2)) * subRefract * 2.5;
        vec3 rd2 = rd;
        rd2.xz *= rot(sunrot.x);
        vec3 waterCol = sat(spc(SPEC - 0.1, WATER_COLOR)) * WATER_TINT
            * (0.4 * pow(min(p.y * 0.7 + 0.9, 1.8), 4.0) * length(skyrd) * (rd2.z * 0.15 + 0.85));
        col += waterCol * 0.17;

        col = mix(col, skyrd, dO / MDIST);
    }
    else{
        col += skyrd;
    }
    col = sat(col);
    col = pow(col, vec3(0.87));
    col *= 1.0 - 0.8 * pow(length(uv * vec2(0.8, 1.0)), 2.7);
    fragColor = vec4(col, 1.0);
}

void mainImage(out vec4 fragColor, in vec2 fragCoord)
{
    if (AA > 1.0) {
        vec4 col = vec4(0.0);
        float px = 1.0 / AA;
        for (int i = 0; i < AA_INT; i++){
            for (int j = 0; j < AA_INT; j++){
                vec4 col2;
                vec2 coord = vec2(fragCoord.x + px * float(i), fragCoord.y + px * float(j));
                render(col2, coord);
                col.rgb += col2.rgb;
            }
        }
        col /= AA * AA;
        fragColor = vec4(col);
    } else {
        render(fragColor, fragCoord);
    }
}
