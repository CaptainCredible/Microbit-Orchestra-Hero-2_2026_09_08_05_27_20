// Sparks struck off the pointer.
//
// The game spawns them - see BG_SHADER_SPARK_* in config.js for how many,
// how long they live and how thickly they are laid - and hands them over as
// iSpark[]. What is below is only how this fire draws one, and the whole
// trick of it is that a spark is put through the *same* domain warp the
// flames are, so it rides the fire instead of floating in front of it.
const float SPARK_CORE   = 0.00005;  // how tight the bright centre is
const float SPARK_GLOW   = 0.00004;  // how far its halo reaches
const float SPARK_RISE   = 0.055;    // how fast it climbs, in screens a second
const float SPARK_WANDER = 0.35;     // how much each one wobbles off on its own
const vec3  SPARK_TINT   = vec3(1.0, 0.55, 0.18);

// Must match BG_SHADER_SPARKS in config.js - GLSL ES needs a constant loop
// bound, so it cannot simply ask how long the array is.
const int   SPARK_COUNT = 28;

float rand(vec2 n) {
    return fract(sin(cos(dot(n, vec2(12.9898,12.1414)))) * 83758.5453);
}

float noise(vec2 n) {
    const vec2 d = vec2(0.0, 1.0);
    vec2 b = floor(n), f = smoothstep(vec2(0.0), vec2(1.0), fract(n));
    return mix(mix(rand(b), rand(b + d.yx), f.x), mix(rand(b + d.xy), rand(b + d.yy), f.x), f.y);
}

float fbm(vec2 n) {
    float total = 0.0, amplitude = 1.0;
    for (int i = 0; i < 5; i++) {
        total += noise(n) * amplitude;
        n += n * 1.7;
        amplitude *= 0.47;
    }
    return total;
}

// The domain warp the fire is sampled through. Pulled out of mainImage so a
// spark can be put through the very same one.
vec2 fireWarp(vec2 pp, float t) {
    pp += sin(pp.yx * 4.0 + vec2(.2, -.3) * t) * 0.04;
    pp += sin(pp.yx * 8.0 + vec2(.6, +.1) * t) * 0.01;
    return pp;
}

// The *whole* coordinate pipeline the flames are sampled through: the warp
// plus the scroll. This is what a spark has to ride if it is to sit in the
// same moving frame of reference as the fire. Factored out so the flames and
// the sparks cannot drift apart again.
vec2 fireField(vec2 fragCoord, float t, float dist) {
    vec2 p = fragCoord.xy * dist / iResolution.xx;
    p = fireWarp(p, t);
    p.x -= t / 1.1;
    return p;
}

void mainImage( out vec4 fragColor, in vec2 fragCoord ) {

    const vec3 c1 = vec3(0.5, 0.0, 0.1);
    const vec3 c2 = vec3(0.9, 0.1, 0.0);
    const vec3 c3 = vec3(0.2, 0.1, 0.7);
    const vec3 c4 = vec3(1.0, 0.9, 0.1);
    const vec3 c5 = vec3(0.1);
    const vec3 c6 = vec3(0.9);

    vec2 speed = vec2(0.1, 0.9);
    float shift = 1.327 + sin(iTime * 2.0) / 2.4;
    float alpha = 1.0;

    float dist = 3.5 - sin(iTime * 0.4) / 1.89;

    vec2 uv = fragCoord.xy / iResolution.xy;
    float aspect = iResolution.x / iResolution.y;

    // Flame sample point: same pipeline sparks will use.
    vec2 p = fireField(fragCoord, iTime, dist);

    float q  = fbm(p - iTime * 0.3  + 1.0  * sin(iTime + 0.5) / 2.0);
    float qb = fbm(p - iTime * 0.4  + 0.1  * cos(iTime)       / 2.0);
    float q2 = fbm(p - iTime * 0.44 - 5.0  * cos(iTime)       / 2.0) - 6.0;
    float q3 = fbm(p - iTime * 0.9  - 10.0 * cos(iTime)       / 15.0) - 4.0;
    float q4 = fbm(p - iTime * 1.4  - 20.0 * sin(iTime)       / 14.0) + 2.0;
    q = (q + qb - .4 * q2 - 2.0 * q3 + .6 * q4) / 3.8;
    vec2 r = vec2(fbm(p + q / 2.0 + iTime * speed.x - p.x - p.y),
                  fbm(p + q      - iTime * speed.y));
    vec3 c = mix(c1, c2, fbm(p + r)) + mix(c3, c4, r.x) - mix(c5, c6, r.y);
    vec3 color = vec3(1.0 / (pow(c + 1.61, vec3(4.0))) * cos(shift * fragCoord.y / iResolution.y));

    color = vec3(1.0, .2, .05) / (pow((r.y + r.y) * max(.0, p.y) + 0.1, 4.0));

    // ---- the sparks ---------------------------------------------------------
    //
    // A spark is struck at a point on the screen and then left to the fire.
    // Its position is carried into the field's own space, put through the SAME
    // warp the flames are - fireField, which is fireWarp plus the scroll - and
    // carried back, so it wobbles with whatever the flame beside it is doing.
    // On top of that it climbs, the way an ember does, and wanders a little on
    // its own seed so that two struck together do not travel as a pair.
    //
    // Added before the tone shoulder below, so a bright spark rolls up to
    // white rather than clipping to a hard dot.
    for (int i = 0; i < SPARK_COUNT; i++) {
        float life = iSpark[i].w;
        if (life <= 0.0) continue;

        vec2  at   = iSpark[i].xy;
        float seed = iSpark[i].z;
        float age  = sparkAge(life);

        // Rises, and drifts on its own seed - the drift grows with age, so
        // they fan apart rather than marching in step.
        at.y += age * SPARK_RISE;
        at.x += sin(seed * 31.4 + age * 1.7) * SPARK_WANDER * age * 0.1;

        // UV -> field space, matching how the flame builds p from fragCoord:
        //   p = fragCoord * dist / iResolution.x
        //     = uv * vec2(aspect, 1.0) * dist
        vec2 sp = at * vec2(aspect, 1.0) * dist;
        // Through the flames' own full transform (warp + scroll).
        sp = fireWarp(sp, iTime);
        sp.x -= iTime / 1.1;
        // Back to UV.
        at = sp / (vec2(aspect, 1.0) * dist);

        vec2  d  = (uv - at) * vec2(aspect, 1.0);
        float r2 = dot(d, d);

        // Fades as the square of its life, and flickers on its own seed.
        float flicker = 0.65 + 0.35 * sin(iTime * 9.0 + seed * 62.8);
        float fade    = life * life * flicker;

        color += SPARK_TINT * fade *
                 (SPARK_CORE / (r2 + SPARK_CORE) * 0.7 +
                  SPARK_GLOW / (r2 + SPARK_GLOW * 8.0));
    }
    // -------------------------------------------------------------------------

    color = color / (1.0 + max(vec3(0), color));
    fragColor = vec4(color.x, color.y, color.z, alpha);
}
