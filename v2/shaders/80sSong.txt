#ifndef disable_sound_texture_sampling
    #undef speed
    #define speed 5.
#endif

#define audio_vibration_amplitude .125

// ---- Camera controls -------------------------------------------------------
#define CAM_PITCH  -0.47
#define CAM_YAW    0.0
// ---------------------------------------------------------------------------

// ---- Colour controls -------------------------------------------------------
#define SKY_BACK    vec3(0.4, 0.1, 0.1)
#define SKY_HAZE    vec3(0.1, 0.0, 0.0)
#define SUN_COLOR   (vec3(1.0, 0.8, 0.4) * 0.75)
#define GROUND_TINT vec3(0.1, 0.0, 0.0)
#define EDGE_COLOR  vec3(0.9, 0.0, 0.0)
#define VAPOR_EDGE  vec3(0.4, 0.5, 1.0)
// ---------------------------------------------------------------------------

// ---- Wave controls ---------------------------------------------------------
// A ring expands across the terrain. Its phase jumps on every press, so the
// wave visibly restarts from wherever you clicked. Pressing at the exact same
// pixel twice won't restart it - that would need a press timestamp from the
// wrapper (see the iWave discussion).
#define WAVE_CYCLE        2.0    // seconds for one full expansion
#define WAVE_RADIUS_MAX   25.0   // world units the ring reaches before resetting
#define WAVE_WIDTH        3.0    // ring thickness in world units
#define WAVE_AMPLITUDE    1.0    // how far it lifts the terrain
#define WAVE_ORIGIN_SCALE 20.0   // screen 0..1 -> world units, roughly
// ---------------------------------------------------------------------------

float jTime;

float round_1(float x) { return sign(x) * floor(abs(x) + 0.5); }
vec3  round_3(vec3  x) { return floor(x + 0.5); }

#ifdef disable_sound_texture_sampling
#define textureMirror(a, b) vec4(0)
#else
vec4 textureMirror(sampler2D tex, vec2 c){
    vec2 cf = fract(c);
    return texture2D(tex, mix(cf, 1.0 - cf, mod(floor(c), 2.0)));
}
#endif

float amp(vec2 p){
    return smoothstep(1.0, 8.0, abs(p.x));
}

float pow512(float a){
    a *= a; a *= a; a *= a; a *= a;
    a *= a; a *= a; a *= a; a *= a;
    return a * a;
}

float pow1d5(float a){
    return a * sqrt(a);
}

float hash21(vec2 co){
    return fract(sin(dot(co.xy, vec2(1.9898, 7.233))) * 45758.5433);
}

float hash(vec2 uv){
    float a = amp(uv);
    return (a > 0.0 ?
        a * pow1d5(hash21(uv))
        : 0.0) - (textureMirror(iChannel0, vec2((uv.x * 29.0 + uv.y) * 0.03125, 1.0)).x) * audio_vibration_amplitude;
}

float edgeMin(float dx, vec2 da, vec2 db, vec2 uv){
    uv.x += 5.0;
    vec3 c = fract(round_3(vec3(uv, uv.x + uv.y)) * (vec3(0.0, 1.0, 2.0) + 0.61803398875));
    float a1 = textureMirror(iChannel0, vec2(c.y, 0.0)).x > 0.6 ? 0.15 : 1.0;
    float a2 = textureMirror(iChannel0, vec2(c.x, 0.0)).x > 0.6 ? 0.15 : 1.0;
    float a3 = textureMirror(iChannel0, vec2(c.z, 0.0)).x > 0.6 ? 0.15 : 1.0;
    return min(min((1.0 - dx) * db.y * a3, da.x * a2), da.y * a1);
}

vec2 trinoise(vec2 uv){
    const float sq = 1.22474487139;
    uv.x *= sq;
    uv.y -= 0.5 * uv.x;
    vec2 d = fract(uv);
    uv -= d;

    bool c = dot(d, vec2(1.0)) > 1.0;

    vec2 dd = 1.0 - d;
    vec2 da = c ? dd : d;
    vec2 db = c ? d  : dd;

    float nn = hash(uv + float(c));
    float n2 = hash(uv + vec2(1.0, 0.0));
    float n3 = hash(uv + vec2(0.0, 1.0));

    float nmid = mix(n2, n3, d.y);
    float ns   = mix(nn, c ? n2 : n3, da.y);
    float dx   = da.x / db.y;
    return vec2(mix(ns, nmid, dx), edgeMin(dx, da, db, uv + d));
}

vec2 map(vec3 p){
    vec2 n = trinoise(p.xz);

    float h = p.y - 2.0 * n.x;

    // Press position in 0..1 of the canvas. iMouse.z is signed - positive
    // while held - and its magnitude is the press x in pixels. iMouse.w is
    // the press y in pixels, counted up from the bottom.
    vec2 originUV = vec2(abs(iMouse.z) / iResolution.x,
                         iMouse.w / iResolution.y);
    vec2 origin = (originUV - 0.5) * WAVE_ORIGIN_SCALE;

    // The press y picks the phase offset: one full turn per screen height.
    // Bottom = 0, middle = half a turn, top = one full turn (which wraps
    // back to 0, so the extremes meet).
    float phaseOffset = originUV.y;

    // Expanding ring. fract() wraps the cycle, so it runs from radius 0 out
    // to WAVE_RADIUS_MAX once every WAVE_CYCLE seconds, forever. The press
    // y sets where in that cycle the wave sits.
    float phase = fract(jTime / WAVE_CYCLE + phaseOffset);
    float front = phase * WAVE_RADIUS_MAX;

    float r = length(p.xz - origin);
    float ring = 1.0 - smoothstep(0.0, WAVE_WIDTH, abs(r - front));

    h += ring * WAVE_AMPLITUDE;

    return vec2(h, n.y);
}

vec3 grad(vec3 p){
    const vec2 e = vec2(0.005, 0.0);
    float a = map(p).x;
    return vec3(map(p + e.xyy).x - a,
                map(p + e.yxy).x - a,
                map(p + e.yyx).x - a) / e.x;
}

vec2 intersect(vec3 ro, vec3 rd){
    float d = 0.0, h = 0.0;
    for (int i = 0; i < 500; i++){
        vec3 p = ro + d * rd;
        vec2 s = map(p);
        h = s.x;
        d += h * 0.5;
        if (abs(h) < 0.003 * d)
            return vec2(d, s.y);
        if (d > 150.0 || p.y > 2.0) break;
    }
    return vec2(-1.0);
}

void addsun(vec3 rd, vec3 ld, inout vec3 col){
    float sun = smoothstep(0.21, 0.2, distance(rd, ld));
    if (sun > 0.0){
        float yd = (rd.y - ld.y);
        float a = sin(3.1 * exp(-(yd) * 14.0));
        sun *= smoothstep(-0.8, 0.0, a);
        col = mix(col, SUN_COLOR, sun);
    }
}

float starnoise(vec3 rd){
    float c = 0.0;
    vec3 p = normalize(rd) * 300.0;
    for (int i = 0; i < 4; i++){
        float fi = float(i);
        vec3 q = fract(p) - 0.5;
        vec3 id = floor(p);
        float c2 = smoothstep(0.5, 0.0, length(q));
        c2 *= step(hash21(id.xz / id.y), 0.06 - fi * fi * 0.005);
        c += c2;
        p = p * 0.6 + 0.5 * p * mat3(3.0/5.0, 0.0, 4.0/5.0,
                                     0.0,    1.0, 0.0,
                                    -4.0/5.0, 0.0, 3.0/5.0);
    }
    c *= c;
    float g = dot(sin(rd * 10.512), cos(rd.yzx * 10.512));
    c *= smoothstep(-3.14, -0.9, g) * 0.5 + 0.5 * smoothstep(-0.3, 1.0, g);
    return c * c;
}

vec3 gsky(vec3 rd, vec3 ld, bool mask){
    float haze = exp2(-5.0 * (abs(rd.y) - 0.2 * dot(rd, ld)));

    float st = mask ? (starnoise(rd)) * (1.0 - min(haze, 1.0)) : 0.0;
    vec3 back = SKY_BACK * (1.0 - 0.5 * textureMirror(iChannel0, vec2(0.5 + 0.05 * rd.x / rd.y, 0.0)).x
        * exp2(-0.1 * abs(length(rd.xz) / rd.y))
        * max(sign(rd.y), 0.0));
    #ifdef city
    float x = round_1(rd.x * 30.0);
    float h = hash21(vec2(x - 166.0));
    bool building = (h * h * 0.125 * exp2(-x * x * x * x * 0.0025) > rd.y);
    if (mask && building)
        back *= 0.0, haze = 0.8, mask = mask && !building;
    #endif
    vec3 col = clamp(mix(back, SKY_HAZE, haze) + st, 0.0, 1.0);
    //if (mask) addsun(rd, ld, col);
    return col;
}

mat3 rotX(float a){
    float s = sin(a), c = cos(a);
    return mat3(1.0, 0.0, 0.0,
                0.0,   c,  -s,
                0.0,   s,   c);
}

mat3 rotY(float a){
    float s = sin(a), c = cos(a);
    return mat3(  c, 0.0,  -s,
                0.0, 1.0, 0.0,
                  s, 0.0,   c);
}

void mainImage(out vec4 fragColor, in vec2 fragCoord)
{
    fragColor = vec4(0.0);
    mat3 cam = rotY(CAM_YAW) * rotX(CAM_PITCH);

    #ifdef AA
    for (int ax = 0; ax < AA; ax++){
    for (int ay = 0; ay < AA; ay++){
        vec2 xyi = vec2(float(ax), float(ay)) / float(AA);
    #else
        vec2 xyi = vec2(0.0);
    #endif
    vec2 uv = (2.0 * (fragCoord + xyi) - iResolution.xy) / iResolution.y;

    const float shutter_speed = 0.25;
    float dt = fract(hash21(float(1) * (fragCoord + xyi)) + iTime) * shutter_speed;
    jTime = mod(iTime - dt * iTimeDelta, 4000.0);
    vec3 ro = vec3(0.0, 1.0, (-20000.0 + jTime * speed));

    #ifdef stereo
        ro += stereo * vec3(0.2 * (float(uv.x > 0.0) - 0.5), 0.0, 0.0);
        const float de = 0.9;
        uv.x = uv.x + 0.5 * (uv.x > 0.0 ? -de : de);
        uv *= 2.0;
    #endif

    vec3 rd = normalize(vec3(uv, 4.0 / 3.0));
    rd = cam * rd;

    vec2 i = intersect(ro, rd);
    float d = i.x;

    vec3 ld = normalize(vec3(0.0, 0.125 + 0.05 * sin(0.1 * jTime), 1.0));

    vec3 fog = d > 0.0 ? exp2(-d * vec3(0.14, 0.1, 0.28)) : vec3(0.0);
    vec3 sky = gsky(rd, ld, d < 0.0);

    vec3 p = ro + d * rd;
    vec3 n = normalize(grad(p));

    float diff = dot(n, ld) + 0.1 * n.y;
    vec3 col = GROUND_TINT * diff;

    vec3 rfd = reflect(rd, n);
    vec3 rfcol = gsky(rfd, ld, true);

    col = mix(col, rfcol, 0.05 + 0.95 * pow(max(1.0 + dot(rd, n), 0.0), 5.0));
    #ifdef VAPORWAVE
        col = mix(col, VAPOR_EDGE, smoothstep(0.05, 0.0, i.y));
        col = mix(sky, col, fog);
        col = sqrt(col);
    #else
        col = mix(col, EDGE_COLOR, smoothstep(0.05, 0.0, i.y));
        col = mix(sky, col, fog);
    #endif
    if (d < 0.0)
        d = 1e6;
    d = min(d, 10.0);
    fragColor += vec4(clamp(col, 0.0, 1.0), d < 0.0 ? 0.0 : 0.1 + exp2(-d));
    #ifdef AA
    }
    }
    fragColor /= float(AA * AA);
    #endif
}
