float hash21(vec2 p){
    p = fract(p*vec2(123.34,456.21));
    p += dot(p, p+45.32);
    return fract(p.x*p.y);
}

float sdSphere(vec3 p, vec3 c, float r){
    return length(p-c) - r;
}

float smin(float a, float b, float k){
    float h = clamp(0.5 + 0.5*(b-a)/k, 0.0, 1.0);
    return mix(b,a,h) - k*h*(1.0-h);
}

float map(vec3 p){
    float t = iTime*0.55;
    vec3 p1 = vec3(sin(t*0.9)*1.15, cos(t*1.3)*0.85, sin(t*0.7)*1.0);
    vec3 p2 = vec3(cos(t*1.1)*1.0,  sin(t*0.8)*1.0,  cos(t*1.4)*0.9);
    vec3 p3 = vec3(sin(t*1.4+2.0)*0.9, cos(t*0.6+1.0)*1.1, sin(t*1.0+3.0)*0.8);
    vec3 p4 = vec3(cos(t*0.7+4.0)*1.2, sin(t*1.2+2.0)*0.7, cos(t*0.9+1.0)*1.0);
    vec3 p5 = vec3(sin(t*1.0+1.0)*0.6, cos(t*1.5+3.0)*0.6, sin(t*1.3+2.0)*1.1);

    float d = sdSphere(p, p1, 0.85);
    d = smin(d, sdSphere(p, p2, 0.75), 0.6);
    d = smin(d, sdSphere(p, p3, 0.65), 0.6);
    d = smin(d, sdSphere(p, p4, 0.55), 0.55);
    d = smin(d, sdSphere(p, p5, 0.5), 0.5);
    return d;
}

vec3 calcNormal(vec3 p){
    vec2 e = vec2(0.001, 0.0);
    return normalize(vec3(
        map(p+e.xyy) - map(p-e.xyy),
        map(p+e.yxy) - map(p-e.yxy),
        map(p+e.yyx) - map(p-e.yyx)
    ));
}

vec3 refractSafe(vec3 I, vec3 N, float eta){
    float cosi = clamp(dot(-I, N), -1.0, 1.0);
    vec3 n = N;
    float e = eta;
    if(cosi < 0.0){
        cosi = -cosi;
        n = -N;
        e = 1.0/eta;
    }
    float k = 1.0 - e*e*(1.0 - cosi*cosi);
    if(k < 0.0) return vec3(0.0);
    return e*I + (e*cosi - sqrt(k))*n;
}

float fresnelSchlick(float cosTheta, float f0){
    return f0 + (1.0-f0)*pow(clamp(1.0-cosTheta,0.0,1.0), 5.0);
}

vec3 bg(vec3 rd){
    float g = rd.y*0.5 + 0.5;
    vec3 top = vec3(0.62,0.62,0.64);
    vec3 bottom = vec3(0.22,0.22,0.24);
    vec3 c = mix(bottom, top, g);
    float glow = pow(max(dot(rd, vec3(0.0,0.0,1.0)), 0.0), 10.0);
    c += glow*vec3(0.08,0.08,0.08);
    return c;
}

void mainImage( out vec4 fragColor, in vec2 fragCoord )
{
    vec2 uv = (fragCoord - 0.5*iResolution.xy)/iResolution.y;

    float yaw   = (iMouse.z > 0.0) ? (iMouse.x/iResolution.x - 0.5)*6.2831853 : iTime*0.15;
    float pitch = (iMouse.z > 0.0) ? (iMouse.y/iResolution.y - 0.5)*3.0        : 0.25;
    pitch = clamp(pitch, -1.2, 1.2);

    vec3 ro = vec3(0.0, 0.0, 4.6);
    mat3 rotY = mat3(
        cos(yaw), 0.0, sin(yaw),
        0.0,      1.0, 0.0,
        -sin(yaw),0.0, cos(yaw)
    );
    mat3 rotX = mat3(
        1.0, 0.0, 0.0,
        0.0, cos(pitch), -sin(pitch),
        0.0, sin(pitch), cos(pitch)
    );
    ro = rotY*rotX*ro;

    vec3 ta = vec3(0.0);
    vec3 ww = normalize(ta - ro);
    vec3 uu = normalize(cross(ww, vec3(0.0,1.0,0.0)));
    vec3 vv = cross(uu, ww);
    vec3 rd = normalize(uv.x*uu + uv.y*vv + 1.7*ww);

    float t = 0.0;
    float dist = 20.0;
    for(int i=0;i<100;i++){
        vec3 p = ro + rd*t;
        float d = map(p);
        if(d < 0.001){ dist = t; break; }
        t += d*0.9;
        if(t > 20.0) break;
    }

    vec3 col;
    if(dist < 20.0){
        vec3 p = ro + rd*dist;
        vec3 n = calcNormal(p);
        float ior = 1.33;
        float cosTheta = clamp(dot(n, -rd), 0.0, 1.0);
        float f0 = pow((1.0-ior)/(1.0+ior), 2.0);
        float fres = fresnelSchlick(cosTheta, f0);

        vec3 reflDir = reflect(rd, n);
        vec3 reflCol = mix(bg(reflDir), vec3(1.0), 0.35 + 0.5*(1.0-cosTheta));

        vec3 lightDir = normalize(vec3(0.6,0.8,0.4));
        float spec = pow(max(dot(reflect(-lightDir, n), -rd), 0.0), 90.0);
        reflCol += vec3(1.0)*spec*2.2;

        vec3 refrDir = refractSafe(rd, n, 1.0/ior);
        vec3 refrCol;
        if(dot(refrDir,refrDir) < 0.25){
            refrCol = reflCol;
        } else {
            vec3 pin = p - n*0.02;
            float tin = 0.0;
            float distIn = -1.0;
            for(int i=0;i<80;i++){
                vec3 q = pin + refrDir*tin;
                float d2 = map(q);
                float ad = abs(d2);
                if(ad < 0.002){ distIn = tin; break; }
                tin += ad*0.9;
                if(tin > 8.0) break;
            }
            if(distIn > 0.0){
                vec3 p2 = pin + refrDir*distIn;
                vec3 n2 = calcNormal(p2);
                vec3 refrDir2 = refractSafe(refrDir, n2, 1.0/ior);
                if(dot(refrDir2,refrDir2) < 0.25){
                    refrCol = bg(reflect(refrDir, n2));
                } else {
                    refrCol = bg(refrDir2);
                }
                float absorb = clamp(distIn*0.12, 0.0, 0.5);
                refrCol = mix(refrCol, vec3(1.0), absorb*0.2);
            } else {
                refrCol = bg(refrDir);
            }
        }

        col = mix(refrCol, reflCol, fres);
    } else {
        col = bg(rd);
    }

    col = col/(col + vec3(1.0));
    col = pow(col, vec3(1.0/2.2));

    vec2 vuv = fragCoord/iResolution.xy;
    float vig = smoothstep(1.0, 0.3, length(vuv - 0.5));
    col *= mix(0.55, 1.0, vig);

    col += (hash21(fragCoord + iTime) - 0.5) * 0.02;

    fragColor = vec4(col, 1.0);
}
