// ---- Camera controls -------------------------------------------------------
#define CAM_PITCH  -0.32
#define CAM_YAW    0.0
// ---------------------------------------------------------------------------

#define GLOW_Y_OFFSET  0.325


// tanh() is not in GLSL ES 1.00.
vec4 tanh4(vec4 x) {
    vec4 xc = clamp(x, -15.0, 15.0);
    vec4 e = exp(2.0 * xc);
    return (e - 1.0) / (e + 1.0);
}

mat3 rotX(float a) {
    float s = sin(a), c = cos(a);
    return mat3(1.0, 0.0, 0.0,
                0.0,   c,  -s,
                0.0,   s,   c);
}

mat3 rotY(float a) {
    float s = sin(a), c = cos(a);
    return mat3(  c, 0.0,  -s,
                0.0, 1.0, 0.0,
                  s, 0.0,   c);
}

#define r(a) mat2(cos(a - 0.0), cos(a - 11.0), \
                  cos(a - 33.0), cos(a - 0.0))

void mainImage(out vec4 o, in vec2 F)
{
    vec2 R = iResolution.xy;

    mat3 cam = rotY(CAM_YAW) * rotX(CAM_PITCH);

    vec2 glowCentre = vec2(0.5, 0.5 + GLOW_Y_OFFSET) * R;

    // Mouse, as a -1..1 range. iMouse.y is in pixels counted up from the
    // bottom, so this is 0 at the middle of the screen, +1 at the top.
    float mouse = (iMouse.y / R.y - 0.5) * 2.0;

    vec3 P, Q;
    float g = 0.0, d = 0.1;

    o = vec4(0.0);

    for (int i = 0; i < 99; i++)
    {
        if (d <= 1e-4) break;

        g += d * 0.3;

        vec3 dir = vec3((F - 0.5 * R) / R.y, 1.0);
        dir = cam * dir;

        P = dir * g;

        P.z += iTime;
        P.xy *= r(P.z * 0.8);

        d = 1.0 - abs(P.y);

        float a = 2.0;
        for (int j = 0; j < 9; j++)
        {
            Q = P * a;
            Q.z += d * 12.5;
            // Mouse shifts the lattice along z, which the inner sum reads as
            // a phase offset - so the whole structure morphs rather than pans.
            //Q.z += mouse * 2.0;
            d -= abs(dot(sin(Q), vec3(1.0))) / (a * 3.0);
            a *= 2.0;
        }

        o += (0.5 + 0.5 * cos(d * 40.0 + P.z * 2.0 + vec4(1.0, 4.4, 4.0, 0.0)))
             / (0.006 + abs(d) * 0.08);
    }

    o = tanh4(o / 2e4 / length((F - glowCentre) / R.y));
}
